cmake_minimum_required(VERSION 3.25)
project(xexinfo CXX C)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF)
set(THIRDPARTY_ROOT ${XENONRECOMP_DIR}/thirdparty)
set(XENONUTILS_ROOT ${XENONRECOMP_DIR}/XenonUtils)
add_subdirectory(${XENONRECOMP_DIR}/thirdparty ${CMAKE_BINARY_DIR}/xr_thirdparty EXCLUDE_FROM_ALL)
add_subdirectory(${XENONRECOMP_DIR}/XenonUtils ${CMAKE_BINARY_DIR}/XenonUtils EXCLUDE_FROM_ALL)
add_executable(xexinfo main.cpp)
target_link_libraries(xexinfo PRIVATE XenonUtils)
